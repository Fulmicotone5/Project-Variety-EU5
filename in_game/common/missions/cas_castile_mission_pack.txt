                    is_owned_or_owned_by_subjects_of = root
                    percent >= 0.7
                }
            }
            region:aridoamerica_region = {
                any_location_in_region = {
                    is_owned_or_owned_by_subjects_of = root
                    percent >= 0.3
                }
            }
        }


        on_completion = {
            IF = {
                limit = {
                    NOT = {
                        location:tenochtitlan = {
                            is_capital = yes
                        }
                    }
                }
                custom_tooltip = {
                    text = cas_new_spain_gets_a_new_capital
                    location:tenochtitlan.owner = {
                        set_capital = location:tenochtitlan
                    }
                    location:tenochtitlan = {
                        change_development = 3
                    }
                }
            }
            
            IF = {
                limit = {
                    NOT = {
                        location:tenochtitlan = { is_market_center = yes }
                    }
                }
                custom_tooltip = {
                    text = cas_new_spains_market_gets_centralized
                    location:tenochtitlan.market = {
                        relocate_market = location:tenochtitlan
                    }
                }
            }
        }
    }
