PLC_f = { 

	level = 4
	required_locations_fraction = 0.5
	rule = historical

		OR = {
			culture = { has_culture_group = culture_group:polish_group }
			culture = { has_culture_group = culture_group:lithuanian_group }
		}
	}
	allow = {		
		trigger_if = {
			limit = { 
				country_exists = c:POL
				this != c:POL
			}
			custom_tooltip = {
				text = PLC_f_trigger_pol
				OR = {
					tag = POL
					NOT = { country_exists = c:POL }
					c:POL = { is_subject_or_below_of = root }
				}			
			}
		}
		trigger_if = {
			limit = { 
				country_exists = c:LIT

					tag = LIT

					c:LIT = { is_subject_or_below_of = root }
				}			
			}
		}
		owns = location:krakow
		owns = location:vilnius
		owns = location:warsaw
	}
	
	name = PLC
	flag = PLC
	adjective = PLC_ADJ
	tag = PLC
	color = map_PLC
	
	areas = {

		greater_poland_area

		prussia_area

		polesia_area

	}
