monthly_country_pulse = {
	effect = {
		ordered_country = {
			limit = {
				is_ai = no
			}
			order_by = monthly_balance
			max = 1
			add_country_modifier = {
				modifier = small_nerf
				months = 1
			}
		}
		if = {
			limit = {
				has_estate_privilege = estate_privilege:PV_loyalty_to_the_popw
				or = {
					country_rank_is_empire = yes
					NOT = {
						religion = religion:catholic
					}
					NOT = {
						country_exists ?= c:PAP
					}
				}
			}
			revoke_estate_privilege = estate_privilege:PV_loyalty_to_the_popw
		}
		if = {
			limit = {
				societal_value:tall_vs_wide > -95
			}
			every_owned_non_rural_location = {
					limit = { has_town_rights = town_rights_type:tall_bonus }
					revoke_town_rights_of_type = town_rights_type:tall_bonus
			}
		}
		if = {
			limit = {
				current_date < 1337.6.20
				current_date > 1337.5.20
				is_ai = yes
				is_subject = no 
				NOT = {
					has_reform = government_reform:daimyo
				}
				total_population > 240
			}
			trigger_event_silently = PV_policies.2
		}
		if = {
            limit = {
				current_date < 1357.7.20
				current_date > 1357.6.20
                is_ai = yes
                country_has_estate = estate_type:nobles_estate
                has_estate_privilege = estate_privilege:consolidated_corruption_of_the_nobles_PV
            }
            revoke_estate_privilege = estate_privilege:consolidated_corruption_of_the_nobles_PV
        }
		if = {
			limit = {
				is_ai = yes
				is_subject = no 
				NOT = {
					has_reform = government_reform:daimyo
				}
				total_population > 400
			}
			add_legitimacy = 2
			add_stability = 2
			add_gold = 10
		}
		if = {
			limit = {
				is_ai = yes
				is_subject = no 
				NOT = {
					has_reform = government_reform:daimyo
				}
				current_age = age_2_renaissance
				total_population > 250
				country_rank_is_county = no
			}
			add_gold = 15
		}
		if = {
			limit = {
				is_ai = yes
				is_subject = no 
				NOT = {
					has_reform = government_reform:daimyo
				}
				current_date > 1367.5.20
				total_population > 400
				country_rank_is_county = no
			}
			add_gold = 25
		}
		if = {
			limit = {
				is_ai = yes
				is_subject = no 
				NOT = {
					has_reform = government_reform:daimyo
				}
				current_date > 1397.5.20
				total_population > 800
				country_rank_is_county = no
			}
			add_gold = 50
		}
		if = {
			limit = {
				is_ai = yes
				is_subject = no 
				NOT = {
					has_reform = government_reform:daimyo
				}
				current_age = age_3_discovery
				total_population > 800
				country_rank_is_county = no
			}
			add_gold = 250
		}
		if = {
			limit = {
				is_ai = yes
				is_subject = no 
				NOT = {
					has_reform = government_reform:daimyo
				}
				current_age = age_4_reformation
				total_population > 800
				country_rank_is_county = no
			}
			add_gold = 759
		}
		if = {
			limit = {
				is_ai = yes
				is_subject = no 
				NOT = {
					has_reform = government_reform:daimyo
				}
				current_age = age_5_absolutism
				total_population > 800
				country_rank_is_county = no
			}
			add_gold = 2000
		}
		if = {
			limit = {
				is_ai = yes
				is_subject = no 
				NOT = {
					has_reform = government_reform:daimyo
				}
				current_age = age_6_revolutions
				total_population > 800
				country_rank_is_county = no
			}
			add_gold = 4000
		}
	}
}
