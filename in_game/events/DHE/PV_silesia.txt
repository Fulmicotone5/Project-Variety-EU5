	fire_only_once = yes

		tag = WRC

		from = 1300.1.1
		to = 1800.1.1
		monthly_chance = 1
	}

	illustration_tags = {
		10 = exterior
		10 = angry
	}
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
	}

	trigger = {
		"estate_power(estate_type:nobles_estate)" >= 0.25
		inflation > 0.1

		name = sileflavor.2.a
		historical_option = yes

			value = 0.05
		}
		add_inflation = inflation_severe_bonus
	}
	option = {
		name = sileflavor.2.b
		add_estate_satisfaction = { type = estate_type:nobles_estate 
			value = -0.15
		}
	}
}

sileflavor.3 = {
	type = country_event
	title = sileflavor.3.title
	desc = sileflavor.3.desc

	fire_only_once = yes
	illustration_tags = {
		10 = exterior
		10 = regular
	}
	immediate = {
		event_illustration_estate_effect = { background = estate_type:clergy_estate foreground = estate_type:clergy_estate }
		random_country = {
						limit = { owns = location:glogow owns = location:zagan owns = location:sulechow }
						save_scope_as = target_country
					}
	}

	dynamic_historical_event = {
		tag = WRC
		tag = SIL
		from = 1300.1.1
		to = 1800.1.1
		monthly_chance = 20
	}
	trigger = {
		location:wroclaw = {
			is_capital = yes
		}
		owns = location:wroclaw
		NOT = {
			owns = location:glogow
		}
		NOT = {
			owns = location:zagan
		}
		NOT = {
			owns = location:sulechow
		}
	}
	

	option = {
		name = sileflavor.3.a
		historical_option = yes
		add_gold = -75
		add_estate_satisfaction = { type = estate_type:clergy_estate 
			value = -0.05
		}
		root = {
			add_casus_belli = {
				target = scope:target_country
				type = casus_belli:cb_conquer_province
				province = location:glogow.province
			}
			add_casus_belli = {
				target = scope:target_country
				type = casus_belli:cb_conquer_province
				province = location:zagan.province
			}
		}
		scope:target_country = { add_opinion = { target = root  modifier = opinion_displeased } }
	}
	option = {
		name = sileflavor.3.b
		add_stability = stability_mild_bonus
		scope:target_country = { add_opinion = { target = root  modifier = opinion_good_relations } }
	}
}

			}
