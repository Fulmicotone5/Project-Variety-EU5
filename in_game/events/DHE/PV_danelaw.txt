daneflavor.2 = {
	type = country_event
	title = daneflavor.2.title
	desc = daneflavor.2.desc

	fire_only_once = yes

	dynamic_historical_event = {
		tag = P01
		from = 1300.1.1
		to = 1800.1.1
		monthly_chance = 100
	}

	illustration_tags = {
		10 = exterior
		10 = happy
	}
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
	}

	trigger = {
		owns = location:lincoln
		owns = location:leicester
		owns = location:derby
		owns = location:nottingham
		owns = location:rutland
	}

	option = {
		name = daneflavor.2.a
		add_prestige = prestige_severe_bonus
		if = {
				limit = { has_reform = government_reform:PV_last_borough }
				remove_reform = government_reform:PV_last_borough
				add_reform = government_reform:PV_five_bur
			}
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
	}
}
