daneflavor.2 = {
	type = country_event
	title = daneflavor.2.title
	desc = daneflavor.2.desc

	fire_only_once = yes

	dynamic_historical_event = {
		tag = P01
		from = 1300.1.1
		to = 1800.1.1
		monthly_chance = 100
	}

	illustration_tags = {
		10 = exterior
		10 = happy
	}
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
	}

	trigger = {
		owns = location:lincoln
		owns = location:leicester
		owns = location:derby
		owns = location:nottingham
		owns = location:rutland
	}

	option = {
		name = daneflavor.2.a
		add_prestige = prestige_severe_bonus
		if = {
				limit = { has_reform = government_reform:PV_last_borough }
				remove_reform = government_reform:PV_last_borough
				add_reform = government_reform:PV_five_bur
			}
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
	}
}

daneflavor.4 = {
	type = country_event
	title = daneflavor.4.title
	desc = daneflavor.4.desc

	fire_only_once = yes

	dynamic_historical_event = {
		tag = P01
		from = 1300.1.1
		to = 1800.1.1
		monthly_chance = 10
	}

	illustration_tags = {
		10 = exterior
		10 = happy
	}
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
	}

	trigger = {
		owns = location:leicester
		location:leicester = {
			NOT = {
				location_rank = location_rank:town
			}
			NOT = {
				location_rank = location_rank:city
			}
		}
	}

	option = {
		name = daneflavor.4.a
		add_gold = -100
		location:leicester = {
			change_location_rank = location_rank:town
			construct_building = {
					building_type = building_type:temple
			}
		}
	}
	option = {
		name = daneflavor.4.b
		add_estate_satisfaction = { type = estate_type:nobles_estate 
			value = -0.05
		}
	}
	option = {
		name = daneflavor.4.c
		add_estate_satisfaction = { type = estate_type:nobles_estate 
			value = -0.15
		}
		add_gold = 300
	}
}

daneflavor.8 = {
	type = country_event
	title = daneflavor.8.title
	desc = daneflavor.8.desc

	fire_only_once = yes

	dynamic_historical_event = {
		tag = P01
		from = 1300.1.1
		to = 1800.1.1
		monthly_chance = 10
	}

	illustration_tags = {
		10 = exterior
		10 = happy
	}
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
	}

	trigger = {
		owns = location:london
		owns = location:lincoln
		location:london = {
			is_capital = no
		}
		location:boston = {
			is_capital = no
		}
	}

	option = {
		name = daneflavor.8.a
		add_estate_satisfaction = { type = estate_type:nobles_estate 
			value = -0.05
		}
		add_estate_satisfaction = { type = estate_type:burghers_estate 
			value = 0.15
		}
	}
	option = {
		name = daneflavor.8.b

		trigger = {
			owns = location:boston
		}
		if = {
			limit = {
				location:boston = {
					NOT = {
						location_rank = location_rank:city
					}
				}
			}
			location:boston = {
				change_location_rank = location_rank:city
			}
		}
		add_gold = -800
		set_capital = location:boston
		change_societal_value = { type = land_vs_naval	value = 25	}
		change_societal_value = { type = belligerent_vs_conciliatory value = -25 }
	}
	option = {
		name = daneflavor.8.c
		set_capital = location:london
		change_societal_value = { type = land_vs_naval	value = 25	}
		change_societal_value = { type = belligerent_vs_conciliatory value = -25 }
	}
}
