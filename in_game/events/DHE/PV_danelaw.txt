daneflavor.2 = {
	type = country_event
	title = daneflavor.2.title
	desc = daneflavor.2.desc

	fire_only_once = yes

	dynamic_historical_event = {
		tag = P01
		from = 1300.1.1
		to = 1800.1.1
		monthly_chance = 100
	}

	illustration_tags = {
		10 = exterior
		10 = happy
	}
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
	}

	trigger = {
		owns = location:lincoln
		owns = location:leicester
		owns = location:derby
		owns = location:nottingham
		owns = location:rutland
	}

	option = {
		name = daneflavor.2.a
		add_prestige = prestige_severe_bonus
		if = {
				limit = { has_reform = government_reform:PV_last_borough }
				remove_reform = government_reform:PV_last_borough
				add_reform = government_reform:PV_five_bur
			}
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
	}
}

		add_estate_satisfaction = { type = estate_type:nobles_estate 
			value = -0.15
		}
		add_gold = 300
	}

daneflavor.8 = {
	type = country_event
	title = daneflavor.8.title
	desc = daneflavor.8.desc

	fire_only_once = yes

	dynamic_historical_event = {
		tag = P01
		from = 1300.1.1
		to = 1800.1.1
		monthly_chance = 10
	}

	illustration_tags = {
		10 = exterior
		10 = happy
	}
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
	}

	trigger = {
		owns = location:london
		owns = location:lincoln
		location:london = {
			is_capital = no
		}
		location:boston = {
			is_capital = no
		}
	}

	option = {
		name = daneflavor.8.a
		add_estate_satisfaction = { type = estate_type:nobles_estate 
			value = -0.05
		}
		add_estate_satisfaction = { type = estate_type:burghers_estate 
			value = 0.15
		}
	}
	option = {
		name = daneflavor.8.b

		trigger = {
			owns = location:boston
		}
		if = {
			limit = {
				location:boston = {
					NOT = {
						location_rank = location_rank:city
					}
				}
			}
			location:boston = {
				change_location_rank = location_rank:city
			}
		}
		add_gold = -800
		set_capital = location:boston
		change_societal_value = { type = land_vs_naval	value = 25	}
		change_societal_value = { type = belligerent_vs_conciliatory value = -25 }
	}
	option = {
		name = daneflavor.8.c
		set_capital = location:london
		change_societal_value = { type = land_vs_naval	value = 25	}
		change_societal_value = { type = belligerent_vs_conciliatory value = -25 }
	}
}

daneflavor.11 = {
	type = country_event
	title = daneflavor.1.title
	desc = daneflavor.11.desc

	fire_only_once = yes

	dynamic_historical_event = {
		tag = P01
		from = 1338.1.1
		to = 1800.1.1
		monthly_chance = 1
	}
	trigger = {
		religion = religion:norse
		country_exists = c:NOL
		location:doncaster = {
			owner = c:NOL
		}
		
	
	}

	illustration_tags = {
        10 = exterior
    }

	immediate = {
		event_illustration_estate_background_effect = { background = estate_type:peasants_estate }

		location:doncaster = {
			save_scope_as = origin_provincepv
		}
		location:lincoln = {
			save_scope_as = target_location2
		}
		
	}

	option = {
		name = daneflavor.11.a
		add_migration = {
			owner = root
			from = scope:origin_provincepv.province_definition
			to = scope:target_location2.province_definition
			religion = religion:norse
			amount = 0.100
			months = 12
		}
		
		c:NOL = { add_opinion = { target = root  modifier = opinion_mildly_displeased } }
	}
}
