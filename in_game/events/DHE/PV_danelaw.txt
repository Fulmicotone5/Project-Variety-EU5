daneflavor.8 = {
	type = country_event
	title = daneflavor.8.title
	desc = daneflavor.8.desc

	fire_only_once = yes

	dynamic_historical_event = {
		tag = P01
		from = 1350.1.1
		to = 1800.1.1
		monthly_chance = 10
	}

	illustration_tags = {
		10 = exterior
		10 = happy
	}
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
	}

	trigger = {
		owns = location:london
		owns = location:lincoln
		location:london = {
			is_capital = no
		}
		location:boston = {
			is_capital = no
		}
	}

	option = {
		name = daneflavor.8.a
		add_estate_satisfaction = { type = estate_type:nobles_estate 
			value = -0.05
		}
		add_estate_satisfaction = { type = estate_type:burghers_estate 
			value = 0.15
		}
	}
	option = {
		name = daneflavor.8.b

		trigger = {
			owns = location:boston
		}
		if = {
			limit = {
				location:boston = {
					NOT = {
						location_rank = location_rank:city
					}
				}
			}
			location:boston = {
				change_location_rank = location_rank:city
			}
		}
		add_gold = -100
		set_capital = location:boston
		change_societal_value = { type = land_vs_naval	value = 25	}
		change_societal_value = { type = belligerent_vs_conciliatory value = -25 }
	}
	option = {
		name = daneflavor.8.c
		set_capital = location:london
		location:london = {
			change_integration_level = integrated
			every_pop = {
				limit = { religion = religion:catholic }
					split_pop = {
					fraction = { 0.4 0.6 }
					religion = root.religion
					culture = root.culture
					}
				}

			}
		change_societal_value = { type = land_vs_naval	value = 10	}
		change_societal_value = { type = belligerent_vs_conciliatory value = -10 }
	}
}
