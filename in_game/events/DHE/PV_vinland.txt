vinflavor.2 = {
	type = country_event
	title = vinflavor.2.title
	desc = vinflavor.2.desc
	fire_only_once = yes
	dynamic_historical_event = {
		tag = P02
		from = 1338.1.1
		to = 1800.1.1
		monthly_chance = 100
	}
	illustration_tags = {
		10 = exterior
		10 = regular
	}
	immediate = {
		if = {
			limit = {
				is_ai = no
			}
			event_illustration_estate_background_effect = {
				background = estate_type:nobles_estate
			}
			scope:from = {
				event_illustration_estate_foreground_effect = {
					foreground = estate_type:nobles_estate
				}
			}
		}
	}
	trigger = {
		owns = location:bacalhieu
		location:bacalhieu = {
			is_capital = yes
		}
	}
	option = {
		name = vinflavor.2.a
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = -0.05
		}
		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = 0.1
		}
	}
	option = {
		name = vinflavor.2.b
		trigger = {
			owns = location:akilasiewakik
		}
		if = {
			limit = {
				location:akilasiewakik = {
					NOT = {
						location_rank = location_rank:city
					}
				}
			}
			location:akilasiewakik = {
				change_location_rank = location_rank:city
			}
		}
		add_gold = -800
		set_capital = location:akilasiewakik
	}
	option = {
		name = vinflavor.2.c
		trigger = {
			owns = location:belle_isle
		}
		set_capital = location:belle_isle
		add_estate_satisfaction = {
			type = estate_type:peasants_estate
			value = 0.15
		}
	}
	option = {
		name = vinflavor.2.d
		trigger = {
			owns = location:uashat
		}
		set_capital = location:uashat
		change_societal_value = {
			type = land_vs_naval
			value = 25
		}
	}
}

vinflavor.3 = {
	type = country_event
	title = vinflavor.3.title
	desc = vinflavor.3.desc
	fire_only_once = yes
	image = "gfx/interface/illustrations/disaster/placeholder.dds"
	dynamic_historical_event = {
		tag = P02
		from = 1338.1.1
		to = 1800.1.1
		monthly_chance = 100
	}
	trigger = {
		country_exists = c:P06
		NOT = {
			c:P06 = {
				is_subject_or_below_of = root
			}
		}
	}
	immediate = {
		c:P06 = {
			save_scope_as = target_country
		}
	}
	option = {
		name = vinflavor.3.a
		c:P02 = {
			add_casus_belli = {
				target = scope:target_country
				type = casus_belli:cb_conquer_province
				province = scope:target_country.capital.province
			}
		}
	}
}
