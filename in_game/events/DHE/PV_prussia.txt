pruflavor.1 = { 
	type = country_event
	title = pruflavor.1.title
	desc = pruflavor.1.desc
	historical_info = pruflavor.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = ANS
		tag = PRU
		tag = F05
		from = 1400.1.1
		to = 1835.1.1
		monthly_chance = 100
	}

	illustration_tags = {
		10 = happy
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
		ruler ?= {
			save_scope_as = target_character
		}
		create_character = {
			first_name = name_frederick_2
			culture = root.culture
			religion = root.religion
			estate = estate_type:nobles_estate
			adm = 25 dip = 25 mil = 25
			dynasty = dynasty:hohenzollern_dynasty
			age = 0
			script = pru_pv_frederick
			save_scope_as = new_heir
		}
		scope:new_heir = {
			move_country = root
			change_father = root.ruler
			change_dynasty = root.ruler.dynasty
			add_trait = trait:industrious
			add_trait = trait:tactical_genius
			add_trait = trait:bold_fighter
			add_trait = trait:administrator
			add_trait = trait:conqueror
			add_trait = trait:child_gifted
		}
	}

	trigger = {
		or = {
			dynasty:hohenzollern_dynasty = {
				any_character_in_dynasty = {
					is_ruler_of = root
				}
			}
			dynasty:hohenzollern_nurnberg_dynasty = {
				any_character_in_dynasty = {
					is_ruler_of = root
				}
			}
		}
		has_or_had_tag = PRU
		NOT = {
			any_character = {
			is_child_of = root.ruler
			}
		}
		ruler ?= {
			is_married = yes
		}
		is_in_union = no
	}

	option = {
		name = pruflavor.1.a
		add_stability = stability_mild_bonus
		scope:new_heir = {
			add_trait = trait:child_gifted
		}
	}

}

pruflavor.3 = {
	type = country_event
	title = pruflavor.3.title
	desc = pruflavor.3.desc

	fire_only_once = yes
	illustration_tags = {
		10 = interior
		10 = angry
	}
	immediate = {
		event_illustration_estate_effect = { background = estate_type:nobles_estate foreground = estate_type:nobles_estate }
		random_character = {
						limit = {
							has_estate = estate_type:nobles_estate
							is_child = no
							is_alive = yes
							is_female = no
						}

					}
		random_country = {
			limit = { owns = location:freiberg }
			save_scope_as = target_country
		}
	}

	dynamic_historical_event = {
		tag = ANS
		tag = FKN
		from = 1300.1.1
		to = 1800.1.1
		monthly_chance = 100
	}
	trigger = {
		NOT = {
			country_exists = c:LDB
		}
		NOT = {
			owns = location:zwickau
		}
		NOT = {
			owns = location:freiberg
		}
		NOT = {
			owns = location:sayda
		}
		NOT = {
			owns = location:chemnitz
		}
	}
	

	option = {
		name = pruflavor.3.a
		historical_option = yes
		add_estate_satisfaction = { type = estate_type:nobles_estate 
			value = 0.05
		}
		add_legitimacy = -5
		root = {
			add_casus_belli = {
				target = scope:target_country
				type = casus_belli:cb_conquer_province
				province = location:freiberg.province
			}
		}
		scope:target_country = { add_opinion = { target = root  modifier = opinion_displeased } }
	}
	option = {
		name = pruflavor.3.b
		add_legitimacy = 20
