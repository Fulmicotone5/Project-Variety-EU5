namespace = PV_policies
PV_policies.1 = {
	type = country_event
	title = PV_policies.1.t
	desc = PV_policies.1.d
	hidden = yes

	trigger = {
		is_active_parliament = yes
	}

	immediate = {
		if = {
			limit = {
				has_law = distribution_of_power_law
				NOT = { has_policy = dop_favor_the_ruler }
				policy:dop_favor_the_ruler = {
					is_available_for = root
				}
			}
			add_policy = policy:dop_favor_the_ruler
			change_parliament_issue_support = {
				value = 0.5
			}
		}
	}
}
